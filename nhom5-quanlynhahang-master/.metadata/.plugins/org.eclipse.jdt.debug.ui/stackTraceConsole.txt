private void registerShift(int selectedShiftIndex, String employeeName) {
    // Get the current day of the week for the column
    int dayOfWeekColumn = LocalDate.now().getDayOfWeek().getValue(); // 1 for Monday, 7 for Sunday
    if (dayOfWeekColumn > 7) dayOfWeekColumn = 1; // Reset for weekly cycle

    // Adjust for the 'Ca' column at index 0
    int column = dayOfWeekColumn;

    // Set the employee name at the selected shift row and column
    tableModel.setValueAt(employeeName, selectedShiftIndex, column);
    JOptionPane.showMessageDialog(this, "Đăng ký thành công cho " + employeeName + " vào ca " + (selectedShiftIndex + 1));
}
